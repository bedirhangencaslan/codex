# Renames this fork must keep when a merge would otherwise revert them.
#
# The rename commit (f06d1e476) changed two different kinds of thing under one heading.
# Only one of them belongs here.
#
#   the product name      prose, comments, messages, anything a person reads.
#                         Reverting it costs nothing, so it is NOT listed here and the
#                         merge driver lets it follow upstream. That is deliberate: the
#                         name is what made ~90% of the 2026-09-20 merge's 256 conflicts,
#                         and letting it follow upstream is what makes them stop.
#
#   the runtime identity  names the running program reads: an environment variable it
#                         looks up, a directory it opens, an enum variant or class the
#                         compiler resolves. Reverting one of these does not produce a
#                         differently-worded message, it produces a program that looks in
#                         the wrong place or does not build.
#
# Format, one per line:   <upstream form> => <our form> [| <path prefix>]
# The optional path prefix limits the rule to files under it.
# Lines starting with `#` are comments. Order matters: longest match first.
#
# Verify with: python scripts/merge/verify-map.py
# That script reads the tree and fails if a load-bearing rename is missing here.

# --- environment and on-disk layout -----------------------------------------
# `utils/home-dir` reads this name and no other; there is no CODEX_HOME fallback.
CODEX_HOME => SUFFICE_HOME

# The plugin manifest directory `core-plugins` looks for. Must precede `.codex`.
.codex-plugin => .suffice-plugin

# The project config directory `config/src/loader` looks for, and the directory
# `protocol/src/permissions.rs` keeps the sandbox out of.
.codex => .suffice

# --- identifiers: reverting these does not compile ---------------------------
#
# An identifier needs TWO rules, and forgetting the second is silent. The uses below are
# qualified (`ToolError::Codex`) and easy to match anywhere; the declaration is a bare
# `Codex,` inside an enum and matches far too much, so it is anchored - by the file it
# lives in, and by the line above it where that is still not enough.
#
# This was found by a trial merge: the uses came back as `Suffice`, the declarations
# stayed `Codex`, and nothing said so. It would have failed at the compiler, but only
# after everything else looked finished.
ToolError::Codex => ToolError::Suffice
Product::Codex => Product::Suffice
LoginAppBrand::Codex => LoginAppBrand::Suffice
LoginSuccessPageBrand::Codex => LoginSuccessPageBrand::Suffice

# ...and their declarations.
Codex(CodexErr) => Suffice(CodexErr) | codex-rs/core/src/tools/sandboxing.rs
#[serde(alias = "CODEX")]\n    Codex, => #[serde(alias = "CODEX")]\n    Suffice, | codex-rs/protocol/src/protocol.rs

# ...and `Self::` inside their own impls, which the qualified rules above do not reach: the
# 2026-09-24 merge left `Self::Codex` in `impl Product` against a variant named `Suffice`,
# and verify-names.py did not see it. Neither file declares another enum with a `Codex` variant.
Self::Codex => Self::Suffice | codex-rs/protocol/src/protocol.rs
Self::Codex => Self::Suffice | codex-rs/login/src/success_page.rs

# The two brand enums had no declaration rule at all: on 2026-09-24 `LoginAppBrand`'s uses
# came back as `Suffice` and its declaration stayed `Codex`, which is also the wire value
# (`rename_all = "lowercase"`). Anchored on the enum's own opening lines.
pub enum LoginAppBrand {\n    #[default]\n    Codex, => pub enum LoginAppBrand {\n    #[default]\n    Suffice, | codex-rs/app-server-protocol/src/protocol/v2/account.rs
pub enum LoginSuccessPageBrand {\n    Codex, => pub enum LoginSuccessPageBrand {\n    Suffice, | codex-rs/login/src/success_page.rs

# The Python SDK exports `Suffice` beside `AsyncCodex`, `CodexError` and `CodexRpcError`,
# which kept their names. Scoped to the SDK so a bare `Codex` elsewhere is left alone.
Codex => Suffice | sdk/python/

# --- the package and binary the user installs --------------------------------
@openai/codex => @suffice/cli | codex-cli/

# The binary cargo builds and the one `cargo run` picks. The fold turns every lowercase
# `suffice` into `codex`, and these are quoted strings, not path segments, so nothing above
# restores them: the 2026-09-24 merge came out building `codex.exe`. Quoted, so the package
# name `codex-cli` and the `codex-*` dependencies are left alone.
"codex" => "suffice" | codex-rs/cli/Cargo.toml

# The CLI's own name: clap's `name` and `bin_name`, the usage line, and the name shell
# completions are generated for. The 2026-09-24 merge folded all four to `codex`, so
# `--version` said `codex` and `suffice completion` wrote completions for a command that is
# not installed. The usage line holds a literal `\n`, which a rule here cannot spell (this
# file reads `\n` as a newline), so it is matched in two halves.
name = "codex", => name = "suffice", | codex-rs/cli/src/main.rs
"codex [OPTIONS] [PROMPT] => "suffice [OPTIONS] [PROMPT] | codex-rs/cli/src/main.rs
codex [OPTIONS] <COMMAND> [ARGS]" => suffice [OPTIONS] <COMMAND> [ARGS]" | codex-rs/cli/src/main.rs
let name = "codex"; => let name = "suffice"; | codex-rs/cli/src/main.rs
